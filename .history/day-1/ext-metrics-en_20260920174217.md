kubectl exec -n deepflow deepflow-clickhouse-0 -- clickhouse-client -q "describe ext_metrics.metrics"
جدول أعمدة ext_metrics.metrics (كل الأعمدة الـ 30)

هذا الجدول يجمع فكرتين رأيناهما سابقاً: أعمدة الموارد (مثل event.event) + مصفوفات Key-Value (مثل deepflow_server). التعليقات صينية ومترجمة.

1) الوقت
#	اسم العمود	النوع	الشرح الأصلي	التوضيح بالعربي
1	time	DateTime	v7.1.7.3	وقت تسجيل المقياس بالثواني (والرقم إصدار البنية)
2) عنوان IP
#	اسم العمود	النوع	الشرح الأصلي	التوضيح بالعربي
2	ip4	IPv4	IPv4地址 = عنوان IPv4	عنوان IPv4 لمصدر المقياس
3	ip6	IPv6	IPV6地址 = عنوان IPv6	عنوان IPv6 لمصدر المقياس
4	is_ipv4	UInt8	是否IPV4地址. 0: 否, ip6 صالح, 1: نعم, ip4 صالح	هل العنوان IPv4 (1) أم IPv6 (0)
3) موارد البنية التحتية (معرّفات رقمية)
#	اسم العمود	النوع	الشرح الأصلي	التوضيح بالعربي
5	region_id	UInt16	ip对应的云平台区域ID = معرّف المنطقة المقابل للـ IP	معرّف المنطقة
6	az_id	UInt16	可用区ID = معرّف منطقة التوافر	معرّف منطقة التوافر
7	host_id	UInt16	宿主机ID = معرّف الخادم المضيف	معرّف الخادم المضيف
8	l3_epc_id	Int32	ip对应的EPC ID	معرّف الشبكة الافتراضية (VPC)
9	l3_device_type	UInt8	ip对应的资源类型 = نوع المورد المقابل للـ IP	نوع المورد
10	l3_device_id	UInt32	ip对应的资源ID = معرّف المورد المقابل للـ IP	معرّف المورد
11	subnet_id	UInt16	ip对应的子网ID(0: 未找到) = معرّف الشبكة الفرعية (0: غير موجود)	معرّف الشبكة الفرعية
12	gprocess_id	UInt32	全局进程ID = معرّف العملية العام	معرّف العملية العام
13	agent_id	UInt16	采集器的ID = معرّف المجمِّع	معرّف وكيل DeepFlow (Agent)
4) موارد Kubernetes
#	اسم العمود	النوع	الشرح الأصلي	التوضيح بالعربي
14	pod_cluster_id	UInt16	容器集群ID = معرّف عنقود الحاويات	معرّف الـ Cluster
15	pod_node_id	UInt32	容器节点ID = معرّف عقدة الحاويات	معرّف عقدة Kubernetes
16	pod_ns_id	UInt16	容器命名空间ID = معرّف الـ Namespace	معرّف الـ Namespace
17	pod_group_id	UInt32	容器工作负载ID = معرّف الـ Workload	معرّف الـ Workload (مثل Deployment)
18	pod_id	UInt32	容器POD ID = معرّف الـ Pod	معرّف الـ Pod
19	service_id	UInt32	ip对应的服务ID = معرّف الخدمة	معرّف الخدمة (Service)
5) الحقول التلقائية (Auto)
#	اسم العمود	النوع	الشرح الأصلي	التوضيح بالعربي
20	auto_instance_id	UInt32	المورد المفضَّل: الـ Pod أولاً. الأولوية: pod_id ← pod_node_id ← l3_device_id	معرّف المثيل، يختار DeepFlow أول قيمة موجودة بهذا الترتيب
21	auto_instance_type	UInt8	نوع المورد: 0 = IP بلا مورد، 0-100 = deviceType (10 = pod، 14 = podNode)، 101-200 = موارد مجرّدة (101 = podGroup، 102 = service)، 201-255 = أخرى	نوع المثيل، وهذا الجدول يشرح رموز القيم
22	auto_service_id	UInt32	المورد المفضَّل: الخدمة أولاً. الأولوية: service_id ← pod_node_id ← l3_device_id	معرّف الخدمة، ترتيب الأولوية مختلف عن auto_instance_id
23	auto_service_type	UInt8	نفس رموز النوع السابقة	نوع الخدمة
6) المقاييس والوسوم (Key-Value)
#	اسم العمود	النوع	الشرح الأصلي	التوضيح بالعربي
24	virtual_table_name	String	虚拟表名 = اسم الجدول الافتراضي	اسم المصدر أو الفئة التي جاءت منها المقاييس
25	team_id	UInt16	团队ID = معرّف الفريق	معرّف الفريق
26	tag_names	Array(String)	额外的tag = وسوم إضافية	أسماء الوسوم
27	tag_values	Array(String)	额外的tag对应的值 = قيم الوسوم	قيم الوسوم (تقابل الأسماء بالترتيب)
28	metrics_float_names	Array(String)	额外的float类型metrics = مقاييس float إضافية	أسماء المقاييس الرقمية
29	metrics_float_values	Array(Float64)	额外的float metrics值 = قيم المقاييس	قيم المقاييس (تقابل الأسماء بالترتيب)
